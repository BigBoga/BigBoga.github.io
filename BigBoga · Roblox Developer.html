-- Player data with ProfileService
local Store = ProfileService.GetProfileStore(
  "PlayerData", Template
)

function DataService:OnJoin(player)
  local profile = Store:LoadProfileAsync(
    "Player_" .. player.UserId
  )
  if profile then
    profile:ListenToRelease(function()
      player:Kick()
    end)
  end
end
